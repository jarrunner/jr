package littlejlib.jr;

import static littlejlib.jr.N.*;

/**
 * macOS variant of ExeInfo - swap in for the Linux ExeInfo.java when building against macOS
 * bindings (PosixApi generated from jr-posix-macos.h / posix-macos.symbols, which is where
 * _NSGetExecutablePath is bound). macOS has no /proc, so it needs Darwin's own API instead of
 * Linux's realpath("/proc/self/exe") trick.
 *
 * NOT WIRED IN YET - kept as .java.txt, not .java, so it is not compiled by build-posix.sh's
 * `find src/main/java-posix -name '*.java'` alongside the Linux ExeInfo.java of the same class
 * name (a duplicate class in one compile would just be a build error, not a real target choice -
 * a genuine per-target source-selection mechanism, matching how ../posix/gen/<target> is already
 * chosen at build time for the demo, is PRP-21 follow-up work once there is a real macOS build
 * script to select it. For now: rename this to ExeInfo.java and delete/rename the Linux one aside
 * when building for macOS by hand.
 *
 * _NSGetExecutablePath(char *buf, uint32_t *bufsize) returns 0 on success and fills buf; if buf is
 * too small it returns -1 and sets *bufsize to the required size - handled below with one retry at
 * the reported size. The path it returns may itself contain symlinks (unlike Linux's
 * /proc/self/exe, which the kernel already resolves), so realpath() on the result matches what
 * ExeInfo promises callers (the resolved path, per the Windows GetFinalPathNameByHandleA-based
 * behaviour the config-sibling-file lookup relies on).
 */
public final class ExeInfo {
    private ExeInfo() {}

    public static String fullPathNoExt() {
        return fullPath();
    }

    public static String baseNameNoExt() {
        return Paths.fileNameOf(fullPath());
    }

    public static String fullPath() {
        var sizeVar = intVar();
        sizeVar.putInt(PosixApi.PATH_MAX);
        var buf = alloc(PosixApi.PATH_MAX);
        var rc = PosixApi.nsGetExecutablePath(buf, sizeVar);
        if (rc != 0) {
            // buffer was too small; sizeVar now holds the required size - retry once
            var needed = sizeVar.getInt();
            buf = alloc(needed);
            if (PosixApi.nsGetExecutablePath(buf, sizeVar) != 0) {
                return "";
            }
        }
        var raw = string(buf);
        var resolved = PosixApi.realpath(cstr(raw), alloc(PosixApi.PATH_MAX));
        return resolved.toLong() == 0 ? raw : string(resolved);
    }
}
